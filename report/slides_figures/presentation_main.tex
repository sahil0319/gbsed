% ================= Slide 2: Introduction =================
\begin{frame}{Introduction}

\begin{itemize}

    \item \textbf{Connected and Autonomous Vehicles (CAVs)} continuously exchange
    information about their \textbf{road environment} for driving.

    \item This includes \textbf{road scenes, nearby vehicles, pedestrians, and road conditions},
    resulting in large amounts of data.

    \item Continuously transmitting this data creates a significant
    \textbf{bandwidth and communication burden}.

    \item Real vehicular channels also experience \textbf{noise, fading, interference,
    and packet loss}, making reliable communication challenging.

\end{itemize}

\vspace{0.45cm}

\begin{center}
    \textit{How can vehicles communicate the most important information
    efficiently and reliably?}
\end{center}

\end{frame}

% SLIDE — RELATED WORK

\begin{frame}{Key Related Works}

\begin{itemize}

    \item \textbf{Scene-Graph Augmented Risk Assessment \cite{yu2020scene}}
    \begin{itemize}
        \item Represents driving scenes as graphs of objects and relationships.
        \item Uses GNN/LSTM models for driving risk assessment.
        \item Establishes structured scene representation for autonomous driving.
    \end{itemize}

    \vspace{0.12cm}

    \item \textbf{SEECAD: Semantic Communication for CAVs \cite{ribouh2024seecad}}
    \begin{itemize}
        \item Transmits semantic information instead of raw image pixels.
        \item Uses MobileNetV2 with LDPC coding and QAM modulation.
        \item Shows semantic communication can reduce transmission requirements.
    \end{itemize}

    \vspace{0.12cm}

    \item \textbf{Knowledge Graph Representation Learning 
    \cite{hello2024knowledge}}
    \begin{itemize}
        \item Represents knowledge using compact graph embeddings.
        \item Reconstructs graph relations at the receiver.
        \item Enables communication-efficient graph transmission.
    \end{itemize}

\end{itemize}

\end{frame}


% ============================================================
% SLIDE — GBSED
% ============================================================
\begin{frame}{GBSED: From Semantic Representation to Risk Assessment}

\begin{itemize}

    \item \textbf{GBSED \cite{ribouh2026gbsed}}
    \begin{itemize}
        \item Encodes driving scene graphs into a compact tensor
        representation for semantic transmission.
        \item Applies relation-based semantic compression to remove
        redundant graph information.
        \item Reconstructs the task-relevant scene graph and uses a
        \textbf{GNN + LSTM + MLP} for downstream risk assessment.
    \end{itemize}

    \vspace{0.15cm}

    \item \textbf{Reported performance}
    \begin{itemize}
        \item Approximately \textbf{99.9\% data reduction}.
        \item Semantic fidelity above \textbf{0.9} at SNR $>10$ dB.
    \end{itemize}

    \vspace{0.15cm}

    \item \textbf{Evaluation}
    \begin{itemize}
        \item MIMO-OFDM channel with 3GPP CDL using \textbf{Sionna}.
        \item Focuses on physical-layer semantic transmission rather than
        full network-level vehicular communication.
    \end{itemize}

\end{itemize}

\end{frame}

% SLIDE 10 — PROCESSING
% ============================================================
\begin{frame}{Methodology: Processing Stages}

\begin{enumerate}

    \item \textbf{Scene Understanding}

    Road image $\rightarrow$ Scene graph

    \item \textbf{Semantic Representation}

    Scene graph $\rightarrow \left(T,\;F\right)$

    \scriptsize
    $T$: relation tensor, $F$: node features

    \normalsize
    \item \textbf{Semantic Compression}

    Remove relation types absent from the scene

    \item \textbf{Serialization}

    Compressed representation $\rightarrow$ Float16 bytes

    \item \textbf{Network Transmission}

    Bytes $\rightarrow$ chunks $\rightarrow$ Base64 $\rightarrow$
    IEEE 802.11p

    \item \textbf{Reconstruction}

    Received bytes $\rightarrow$ deserialization $\rightarrow$
    semantic decompression $\rightarrow$ scene graph

\end{enumerate}

\end{frame}

\begin{frame}{Why Does Semantic Communication Work Here?}

\begin{center}
{\Large \textbf{The task does not need the whole image.}}

\vspace{0.35cm}


\noindent
\textbf{\large Why GBSED fits this task}

\vspace{0.2cm}

\begin{columns}[T]

\column{0.32\textwidth}

\textbf{\large 1. Compress the Right Information}

\vspace{0.12cm}

The image contains much more information
than the risk-assessment task needs.

\vspace{0.12cm}

GBSED keeps the information that matters:

\begin{itemize}
\item Objects
\item Positions
\item Relations
\end{itemize}

\column{0.32\textwidth}

\textbf{\large 2. Degrade Differently}

\vspace{0.12cm}

Image compression progressively loses
visual detail.

\vspace{0.12cm}

A scene graph can lose individual
relation types while preserving the
remaining semantic structure.

\column{0.32\textwidth}

\textbf{\large 3. Meaning Is Discrete}

\vspace{0.12cm}

A relation is essentially:

\[
\text{present / absent}
\]

A label is a small integer.

\vspace{0.12cm}

Valid semantic content can therefore
remain exact even with limited bandwidth.

\end{columns}

\vspace{0.35cm}

\noindent
\rule{\textwidth}{0.5pt}



\end{center}

\end{frame}

% ============================================================
% SLIDE 21 — LIMITATIONS
% ============================================================
\begin{frame}{Limitations}

\begin{itemize}

    \item Both driving sequences were labelled as risky, so we
          could not compare risky and safe cases.

    \vspace{0.15cm}

    \item Only two driving sequences and two vehicles were tested.

    \vspace{0.15cm}

    \item Only two vehicles were used, so there was very little
          network traffic.

    \vspace{0.15cm}

    \item The risk labels were based on simple rules.

    \vspace{0.15cm}

    \item The slice-aligned payload helps most when a frame spans several
          chunks; with larger chunks or sparser scenes a single chunk already
          carries the whole frame, so there is nothing to lose partially.

    \vspace{0.15cm}

    \item More vehicle movement and network conditions need to
          be tested to better understand the communication range.

\end{itemize}

\end{frame}
% ============================================================
% SLIDE 22 — CONCLUSION
% ============================================================
\begin{frame}{Conclusion}

\begin{center}
\Large
\textbf{What We Demonstrated}
\end{center}

\vspace{0.25cm}

\begin{itemize}

    \item Integrated GBSED with \textbf{Veins, OMNeT++, SUMO}
          and IEEE 802.11p for vehicular communication.

    \item Successfully reconstructed semantic payloads when
          network frames were delivered.

    \item At the \textbf{same byte budget}, GBSED preserved
          safety-critical relations while JPEG/WebP did not.

    \item WebP needed approximately \textbf{96$\times$ more bandwidth}
          to achieve the same safety-relation preservation.

    \item The payload is packed so that semantic information degrades
          \textbf{by relation content} --- a lost packet costs a relation
          type, not the whole frame.

\end{itemize}



\end{frame}
